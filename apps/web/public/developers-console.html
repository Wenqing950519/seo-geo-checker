import { GeoCheckClient } from '@geocheck/sdk';

const client = new GeoCheckClient({ apiKey: 'YOUR_API_KEY' });
const job = await client.measurements.create({
  url: 'https://example.com',
  query: '台北高檔原木牛排約會餐廳推薦',
  engines: ['openai', 'gemini', 'perplexity', 'anthropic']
});

console.log('觀測任務完成:', job.results);
